DROP TABLE IF EXISTS tareas_usuarios CASCADE;
DROP TABLE IF EXISTS tareas CASCADE;
DROP TABLE IF EXISTS equipos CASCADE;
DROP TABLE IF EXISTS usuarios CASCADE;

-- =========================
-- TABLA USUARIOS
-- =========================

CREATE TABLE usuarios (
    id_usuario INT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(100) NOT NULL,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =========================
-- TABLA EQUIPOS
-- =========================

CREATE TABLE equipos (
    id_equipo INT PRIMARY KEY,
    nombre_equipo VARCHAR(100) NOT NULL,
    descripcion TEXT
);

-- =========================
-- TABLA TAREAS
-- =========================

CREATE TABLE tareas (
    id_tarea INT PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    descripcion TEXT,
    estado VARCHAR(50) DEFAULT 'Pendiente',
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    id_equipo INT REFERENCES equipos(id_equipo)
);

-- =========================
-- TABLA INTERMEDIA
-- MUCHOS A MUCHOS
-- =========================

CREATE TABLE tareas_usuarios (
    id_tarea INT REFERENCES tareas(id_tarea),
    id_usuario INT REFERENCES usuarios(id_usuario),

    PRIMARY KEY(id_tarea, id_usuario)
);

SELECT
    t.id_tarea,
    t.titulo,
    t.descripcion,
    t.estado,
    t.fecha_creacion,

    u.id_usuario,
    u.nombre,
    u.email,
    u.password,
    u.fecha_creacion AS fecha_usuario,

    e.id_equipo,
    e.nombre_equipo,
    e.descripcion AS descripcion_equipo

FROM tareas_usuarios tu
JOIN tareas t
    ON tu.id_tarea = t.id_tarea
JOIN usuarios u
    ON tu.id_usuario = u.id_usuario
JOIN equipos e
    ON t.id_equipo = e.id_equipo

ORDER BY t.id_tarea, u.id_usuario;
